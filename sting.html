Q14 Enter the string from user and enter a character,it is exist or not? Answer: void main() { char str[10],ch; int x,count=0,i; printf("Enter string "); scanf("%s",str); x=strlen(str); printf("Enter character "); scanf("%s",&ch); for(i=0;i<x;i++) { if(str[i]==ch) { count++; printf("It exists "); break; } } if(count==0) { printf("it doesn't exist"); } }
